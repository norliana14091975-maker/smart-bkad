   107→   107→export function AdminSettingsSection() {
   108→   108→  const { toast } = useToast()

   110→   110→  const { data, isLoading } = useSettings()
   111→   111→
   112→   112→  // draf perubahan teks; nilai efektif = draf > data > bawaan
   113→   113→  const [draft, setDraft] = useState<Partial<Record<TextKey, string>>>({})
   114→   114→  const [saving, setSaving] = useState(false)
   115→   115→  const [resetOpen, setResetOpen] = useState(false)
   116→   116→  const [uploading, setUploading] = useState<'logo' | 'sidebar-logo' | 'emblem' | 'favicon' | null>(null)
   117→   117→
   118→   118→  const logoInputRef = useRef<HTMLInputElement>(null)
   119→   119→  const sidebarLogoInputRef = useRef<HTMLInputElement>(null)

   122→   122→
   123→   123→  // Draf warna header (hex); null = belum diubah (ikut data server)
   124→   124→  const [headerColorDraft, setHeaderColorDraft] = useState<string | null>(null)
   125→   125→  const headerColorValue = headerColorDraft ?? (data?.headerColor ?? '')
   126→   126→
   127→   127→  // Simpan warna header lewat endpoint yang sama dengan teks
   128→   128→  async function saveHeaderColor(value: string) {
   129→   129→    const color = value.trim()
   130→   130→    if (color !== '' && !/^#[0-9a-fA-F]{6}$/.test(color)) {
   131→   131→      toast({ title: 'Warna tidak valid', description: 'Gunakan kode hex #rrggbb.', variant: 'destructive' })
   132→   132→      return
   133→   133→    }
   134→   134→    try {
   135→   135→      const res = await fetch('/api/admin/settings', {
   136→   136→        method: 'PUT',
   137→   137→        headers: { 'Content-Type': 'application/json' },
   138→   138→        body: JSON.stringify({ headerColor: color }),
   139→   139→      })
   140→   140→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   141→   141→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan warna')
   142→   142→      setHeaderColorDraft(null)
   143→   143→      toast({ title: color ? 'Warna header diperbarui' : 'Warna header kembali ke gradien bawaan' })
   144→   144→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   145→   145→    } catch (err) {
   146→   146→      toast({ title: 'Gagal menyimpan warna', description: String(err), variant: 'destructive' })
   147→   147→    }
   148→   148→  }
   149→   149→
   150→   150→  const value = (key: TextKey): string => {
   151→   151→    if (draft[key] !== undefined) return draft[key]
   152→   152→    const current = data?.[key]
   153→   153→    if (typeof current === 'string') return current
   154→   154→    return DEFAULT_SETTINGS[key]
   155→   155→  }
   156→   156→
   157→   157→  const dirty = TEXT_FIELDS.some((f) => draft[f.key] !== undefined)
   158→   158→
   159→   159→  async function handleSave() {
   160→   160→    if (saving) return
   161→   161→    setSaving(true)
   162→   162→    try {
   163→   163→      const body: Record<string, string> = {}
   164→   164→      for (const f of TEXT_FIELDS) body[f.key] = value(f.key)
   165→   165→      const res = await fetch('/api/admin/settings', {
   166→   166→        method: 'PUT',
   167→   167→        headers: { 'Content-Type': 'application/json' },
   168→   168→        body: JSON.stringify(body),
   169→   169→      })
   170→   170→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   171→   171→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan')
   172→   172→      toast({ title: 'Pengaturan tersimpan' })
   173→   173→      setDraft({})
   174→   174→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   175→   175→    } catch (err) {
   176→   176→      toast({ title: 'Gagal menyimpan', description: String(err), variant: 'destructive' })
   177→   177→    } finally {
   178→   178→      setSaving(false)
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  async function uploadImage(kind: 'logo' | 'sidebar-logo' | 'emblem' | 'favicon', file: File) {
   183→   183→    const ref =
   184→   184→      kind === 'logo'
   185→   185→        ? logoInputRef
   186→   186→        : kind === 'sidebar-logo'
   187→   187→          ? sidebarLogoInputRef
   188→   188→          : kind === 'emblem'
   189→   189→            ? emblemInputRef
   190→   190→            : faviconInputRef
   191→   191→    if (ref) {
   192→   192→      // reset nilai input agar file bernama sama bisa diunggah ulang
   193→   193→      if (ref.current) ref.current.value = ''
   194→   194→    }
   195→   195→    if (!file.type.startsWith('image/')) {
   196→   196→      toast({ title: 'File bukan gambar', description: 'Pilih file PNG/JPG/GIF/WebP/SVG/ICO.', variant: 'destructive' })
   197→   197→      return
   198→   198→    }
   199→   199→    if (file.size > 2 * 1024 * 1024) {
   200→   200→      toast({ title: 'File terlalu besar', description: 'Ukuran maksimum 2 MB.', variant: 'destructive' })
   201→   201→      return
   202→   202→    }
   203→   203→    setUploading(kind)
   204→   204→    try {
   205→   205→      const fd = new FormData()
   206→   206→      fd.append('file', file)
   207→   207→      const res = await fetch(`/api/admin/settings/${kind}`, { method: 'POST', body: fd })
   208→   208→      const json = (await res.json()) as {
   209→   209→        data?: { logoUrl?: string; sidebarLogoUrl?: string; emblemUrl?: string; faviconUrl?: string }
   210→   210→        error?: string
   211→   211→      }
   212→   212→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal mengunggah')
   213→   213→      toast({
   214→   214→        title:
   215→   215→          kind === 'logo'
   216→   216→            ? 'Logo diperbarui'
   217→   217→            : kind === 'sidebar-logo'
   218→   218→              ? 'Logo pojok kiri diperbarui'
   219→   219→              : kind === 'emblem'
   220→   220→                ? 'Lencana pojok kanan diperbarui'
   221→   221→                : 'Favicon diperbarui',
   222→   222→      })
   223→   223→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   224→   224→    } catch (err) {
   225→   225→      toast({ title: 'Gagal mengunggah', description: String(err), variant: 'destructive' })
   226→   226→    } finally {
   227→   227→      setUploading(null)
   228→   228→    }
   229→   229→  }

   244→   244→      })

   246→   246→    } catch (err) {
   247→   247→      toast({ title: 'Gagal menghapus', description: String(err), variant: 'destructive' })
   248→   248→    }

   310→   310→        <div className="mt-4 flex flex-wrap items-center justify-end gap-2">

   398→   398→        <section className="rounded-lg border bg-card p-4 shadow-sm sm:p-5">
