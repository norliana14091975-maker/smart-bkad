   107→export function AdminSettingsSection() {
   108→  const { toast } = useToast()
   109→  const queryClient = useQueryClient()
   110→  const { data, isLoading } = useSettings()
   111→
   112→  // draf perubahan teks; nilai efektif = draf > data > bawaan
   113→  const [draft, setDraft] = useState<Partial<Record<TextKey, string>>>({})
   114→  const [saving, setSaving] = useState(false)
   115→  const [resetOpen, setResetOpen] = useState(false)
   116→  const [uploading, setUploading] = useState<'logo' | 'sidebar-logo' | 'emblem' | 'favicon' | null>(null)
   117→
   118→  const logoInputRef = useRef<HTMLInputElement>(null)
   119→  const sidebarLogoInputRef = useRef<HTMLInputElement>(null)
   120→  const emblemInputRef = useRef<HTMLInputElement>(null)
   121→  const faviconInputRef = useRef<HTMLInputElement>(null)
   122→
   123→  // Draf warna header (hex); null = belum diubah (ikut data server)
   124→  const [headerColorDraft, setHeaderColorDraft] = useState<string | null>(null)
   125→  const headerColorValue = headerColorDraft ?? (data?.headerColor ?? '')
   126→
   127→  // Simpan warna header lewat endpoint yang sama dengan teks
   128→  async function saveHeaderColor(value: string) {
   129→    const color = value.trim()
   130→    if (color !== '' && !/^#[0-9a-fA-F]{6}$/.test(color)) {
   131→      toast({ title: 'Warna tidak valid', description: 'Gunakan kode hex #rrggbb.', variant: 'destructive' })
   132→      return
   133→    }
   134→    try {
   135→      const res = await fetch('/api/admin/settings', {
   136→        method: 'PUT',
   137→        headers: { 'Content-Type': 'application/json' },
   138→        body: JSON.stringify({ headerColor: color }),
   139→      })
   140→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   141→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan warna')
   142→      setHeaderColorDraft(null)
   143→      toast({ title: color ? 'Warna header diperbarui' : 'Warna header kembali ke gradien bawaan' })
   144→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   145→    } catch (err) {
   146→      toast({ title: 'Gagal menyimpan warna', description: String(err), variant: 'destructive' })
   147→    }
   148→  }
   149→
   150→  const value = (key: TextKey): string => {
   151→    if (draft[key] !== undefined) return draft[key]
   152→    const current = data?.[key]
   153→    if (typeof current === 'string') return current
   154→    return DEFAULT_SETTINGS[key]
   155→  }
   156→
   157→  const dirty = TEXT_FIELDS.some((f) => draft[f.key] !== undefined)
   158→
   159→  async function handleSave() {
   160→    if (saving) return
   161→    setSaving(true)
   162→    try {
   163→      const body: Record<string, string> = {}
   164→      for (const f of TEXT_FIELDS) body[f.key] = value(f.key)
   165→      const res = await fetch('/api/admin/settings', {
   166→        method: 'PUT',
   167→        headers: { 'Content-Type': 'application/json' },
   168→        body: JSON.stringify(body),
   169→      })
   170→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   171→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan')
   172→      toast({ title: 'Pengaturan tersimpan' })
   173→      setDraft({})
   174→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   175→    } catch (err) {
   176→      toast({ title: 'Gagal menyimpan', description: String(err), variant: 'destructive' })
   177→    } finally {
   178→      setSaving(false)
   179→    }
   180→  }
   181→
   182→  async function uploadImage(kind: 'logo' | 'sidebar-logo' | 'emblem' | 'favicon', file: File) {
   183→    const ref =
   184→      kind === 'logo'
   185→        ? logoInputRef
   186→        : kind === 'sidebar-logo'
   187→          ? sidebarLogoInputRef
   188→          : kind === 'emblem'
   189→            ? emblemInputRef
   190→            : faviconInputRef
   191→    if (ref) {
   192→      // reset nilai input agar file bernama sama bisa diunggah ulang
   193→      if (ref.current) ref.current.value = ''
   194→    }
   195→    if (!file.type.startsWith('image/')) {
   196→      toast({ title: 'File bukan gambar', description: 'Pilih file PNG/JPG/GIF/WebP/SVG/ICO.', variant: 'destructive' })
   197→      return
   198→    }
   199→    if (file.size > 2 * 1024 * 1024) {
   200→      toast({ title: 'File terlalu besar', description: 'Ukuran maksimum 2 MB.', variant: 'destructive' })
   201→      return
   202→    }
   203→    setUploading(kind)
   204→    try {
   205→      const fd = new FormData()
   206→      fd.append('file', file)
   207→      const res = await fetch(`/api/admin/settings/${kind}`, { method: 'POST', body: fd })
   208→      const json = (await res.json()) as {
   209→        data?: { logoUrl?: string; sidebarLogoUrl?: string; emblemUrl?: string; faviconUrl?: string }
   210→        error?: string
   211→      }
   212→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal mengunggah')
   213→      toast({
   214→        title:
   215→          kind === 'logo'
   216→            ? 'Logo diperbarui'
   217→            : kind === 'sidebar-logo'
   218→              ? 'Logo pojok kiri diperbarui'
   219→              : kind === 'emblem'
   220→                ? 'Lencana pojok kanan diperbarui'
   221→                : 'Favicon diperbarui',
   222→      })
   223→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   224→    } catch (err) {
   225→      toast({ title: 'Gagal mengunggah', description: String(err), variant: 'destructive' })
   226→    } finally {
   227→      setUploading(null)
   228→    }
   229→  }
   230→
   231→  async function removeImage(kind: 'logo' | 'sidebar-logo' | 'emblem' | 'favicon') {
   232→    try {
   233→      const res = await fetch(`/api/admin/settings/${kind}`, { method: 'DELETE' })
   234→      if (!res.ok) throw new Error('Gagal menghapus')
   235→      toast({
   236→        title:
   237→          kind === 'logo'
   238→            ? 'Logo kustom dihapus'
   239→            : kind === 'sidebar-logo'
   240→              ? 'Logo pojok kiri kembali mengikuti Logo Aplikasi'
   241→              : kind === 'emblem'
   242→                ? 'Lencana pojok kanan kembali ke emblem emas bawaan'
   243→                : 'Favicon kustom dihapus',
   244→      })
   245→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   246→    } catch (err) {
   247→      toast({ title: 'Gagal menghapus', description: String(err), variant: 'destructive' })
   248→    }
   249→  }
   250→
   251→  async function handleReset() {

   259→      toast({ title: 'Gagal mereset', description: String(err), variant: 'destructive' })

   261→  }

   285→                </span>
   286→              </Label>
   287→              {f.textarea ? (
   288→                <Textarea
   289→                  id={`set-${f.key}`}
   290→                  value={value(f.key)}
   291→                  placeholder={f.placeholder}

   293→                  rows={2}
   294→                  onChange={(e) => setDraft((d) => ({ ...d, [f.key]: e.target.value }))}
   295→                />
   296→              ) : (
   297→                <Input
   298→                  id={`set-${f.key}`}
   299→                  value={value(f.key)}
   300→                  placeholder={f.placeholder}
   301→                  maxLength={f.max}
   302→                  onChange={(e) => setDraft((d) => ({ ...d, [f.key]: e.target.value }))}
   303→                />
   304→              )}
   305→              {f.hint && <p className="mt-1 text-[11px] text-muted-foreground">{f.hint}</p>}

   328→                <Save className="h-4 w-4" aria-hidden="true" /> Simpan Pengaturan
   329→              </>
   330→            )}
   331→          </Button>
   332→        </div>
   333→      </section>
   334→
   335→      {/* Logo & Favicon */}
   336→      <div className="mb-5 grid grid-cols-1 gap-5 md:grid-cols-2 xl:grid-cols-3">
   337→        <section className="rounded-lg border bg-card p-4 shadow-sm sm:p-5">
   338→          <h3 className="mb-1 text-sm font-bold uppercase tracking-wide text-foreground/80">
   339→            Logo Aplikasi
   340→          </h3>
   341→          <p className="mb-4 text-xs text-muted-foreground">
   342→            Tampil di pita header (dan jadi fallback logo pojok kiri sidebar).
   343→            PNG/JPG/GIF/WebP/SVG/ICO, maks 2 MB.
   344→          </p>
   345→          <div className="flex items-center gap-4">
   346→            <div className="flex h-20 w-20 shrink-0 items-center justify-center rounded-lg border bg-muted/30 p-2">
   347→              {isLoading ? (
   348→                <Skeleton className="h-14 w-14" />

   352→                <DkiEmblem className="h-14 w-14" />

   357→                {data?.logoUrl ? 'Logo kustom aktif' : 'Menggunakan emblem bawaan'}
   358→              </p>
   359→              <div className="mt-2 flex flex-wrap gap-2">
   360→                <input
   361→                  ref={logoInputRef}
   362→                  type="file"
   363→                  accept="image/*,.ico,.svg"

   365→                  onChange={(e) => {

   369→                />
   370→                <Button
   371→                  variant="outline"
   372→                  size="sm"
   373→                  onClick={() => logoInputRef.current?.click()}
   374→                  disabled={uploading === 'logo'}
   375→                >
   376→                  {uploading === 'logo' ? (

   390→                    <Trash2 className="h-4 w-4" aria-hidden="true" /> Hapus

   393→              </div>
   394→            </div>
   395→          </div>

   398→        <section className="rounded-lg border bg-card p-4 shadow-sm sm:p-5">

   400→            Logo Pojok Kiri (Sidebar)

   403→            Logo kecil di pojok kiri atas aplikasi. Bila kosong, mengikuti Logo Aplikasi.
