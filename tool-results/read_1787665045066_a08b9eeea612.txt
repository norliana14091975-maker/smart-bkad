    94→/** Hook status Setup Wizard (dipakai bersama oleh wizard & kartu di Pengaturan). */
    95→export function useSetupWizardStatus(enabled = true) {
    96→  return useQuery({
    97→    queryKey: ['setup-wizard-status'],
    98→    enabled,
    99→    queryFn: async (): Promise<SetupWizardStatusDto> => {
   100→      const res = await fetch('/api/admin/setup-wizard')
   101→      if (!res.ok) throw new Error('Gagal memuat status Setup Wizard')
   102→      const json = (await res.json()) as { data: SetupWizardStatusDto }
   103→      return json.data
   104→    },
   105→    retry: 1,
   106→  })
   107→}

   114→export function SetupWizard({ open, onOpenChange }: SetupWizardProps) {
   115→  const { toast } = useToast()
   116→  const queryClient = useQueryClient()
   117→  const settingsQuery = useSettings()
   118→  const statusQuery = useSetupWizardStatus(open)
   119→
   120→  const [step, setStep] = useState(0)
   121→
   122→  // ---- Langkah identitas: draf > nilai server ----
   123→  const [idDraft, setIdDraft] = useState<Partial<Record<(typeof IDENTITY_FIELDS)[number]['key'], string>>>({})
   124→  const [idSaving, setIdSaving] = useState(false)
   125→
   126→  const identityValue = (key: (typeof IDENTITY_FIELDS)[number]['key']): string => {
   127→    if (idDraft[key] !== undefined) return idDraft[key]
   128→    const current = settingsQuery.data?.[key]
   129→    return typeof current === 'string' ? current : ''
   130→  }
   131→
   132→  // ---- Langkah keamanan: ganti password akun sendiri ----
   133→  const [pwCurrent, setPwCurrent] = useState('')
   134→  const [pwNew, setPwNew] = useState('')
   135→  const [pwConfirm, setPwConfirm] = useState('')
   136→  const [showPw, setShowPw] = useState(false)
   137→  const [pwSaving, setPwSaving] = useState(false)
   138→  const [pwDone, setPwDone] = useState(false)
   139→  const [pwError, setPwError] = useState<string | null>(null)
   140→
   141→  // ---- Langkah AI Copilot: draf konfigurasi provider ----
   142→  const copilotQuery = useQuery({
   143→    queryKey: ['copilot-settings'],
   144→    enabled: open,
   145→    queryFn: async (): Promise<CopilotSettingsDto> => {
   146→      const res = await fetch('/api/admin/settings/copilot')
   147→      if (!res.ok) throw new Error('Gagal memuat konfigurasi AI Copilot')
   148→      const json = (await res.json()) as { data: CopilotSettingsDto }
   149→      return json.data
   150→    },
   151→    staleTime: 30_000,
   152→    retry: 1,
   153→  })
   154→  const [cpHydrated, setCpHydrated] = useState(false)
   155→  const [cpProvider, setCpProvider] = useState('default')
   156→  const [cpModel, setCpModel] = useState('')
   157→  const [cpBase, setCpBase] = useState('')
   158→  const [cpKey, setCpKey] = useState('')
   159→  const [cpSaving, setCpSaving] = useState(false)
   160→  const [cpTesting, setCpTesting] = useState(false)
   161→  const [cpTest, setCpTest] = useState<{ ok: boolean; message: string } | null>(null)
   162→
   163→  const [finishing, setFinishing] = useState(false)
   164→
   165→  // Reset wizard saat dibuka ulang; bersihkan data sensitif saat ditutup
   166→  useEffect(() => {
   167→    if (open) {
   168→      setStep(0)
   169→    } else {
   170→      setPwCurrent('')
   171→      setPwNew('')
   172→      setPwConfirm('')
   173→      setPwDone(false)
   174→      setPwError(null)
   175→      setCpTest(null)
   176→      setFinishing(false)
   177→    }
   178→  }, [open])
   179→
   180→  // Isi draf copilot dari data server saat pertama dimuat
   181→  useEffect(() => {
   182→    if (copilotQuery.data && !cpHydrated) {
   183→      setCpProvider(copilotQuery.data.provider)
   184→      setCpModel(copilotQuery.data.model ?? '')
   185→      setCpBase(copilotQuery.data.baseUrl ?? '')
   186→      setCpHydrated(true)
   187→    }
   188→  }, [copilotQuery.data, cpHydrated])
   189→
   190→  const cpProviderDef = findCopilotProvider(cpProvider) ?? COPILOT_PROVIDERS[0]
   191→  const cpIsDefault = cpProviderDef.id === 'default'
   192→  const cpKeyPlaceholder = copilotQuery.data?.hasApiKey
   193→    ? `${copilotQuery.data.apiKeyMasked} — tersimpan (biarkan kosong untuk memakainya)`
   194→    : cpProviderDef.keyHint || 'API Key'
   195→
   196→  const status = statusQuery.data
   197→  const checks = status?.checks
   198→
   199→  /** Perbarui status wizard (checklist langkah pengantar & selesai). */
   200→  async function refreshStatus() {
   201→    await queryClient.invalidateQueries({ queryKey: ['setup-wizard-status'] })
   202→  }
   203→
   204→  /** Ganti provider: praisi Base URL & contoh model dari registry. */
   205→  function selectCpProvider(id: string) {
   206→    setCpProvider(id)
   207→    const p = findCopilotProvider(id)
   208→    setCpBase(p?.baseUrl ?? '')
   209→    setCpModel(p?.modelPlaceholder ?? '')
   210→    setCpKey('')
   211→    setCpTest(null)
   212→  }
   213→
   214→  /** Simpan identitas dashboard, lalu lanjut ke langkah berikutnya. */
   215→  async function saveIdentity() {
   216→    if (idSaving) return
   217→    if (!identityValue('appTitle').trim() || !identityValue('govName').trim()) {
   218→      toast({
   219→        title: 'Lengkapi identitas',
   220→        description: 'Judul Dashboard dan Nama Pemerintah Daerah wajib diisi.',
   221→        variant: 'destructive',
   222→      })
   223→      return
   224→    }
   225→    setIdSaving(true)
   226→    try {
   227→      const body: Record<string, string> = {}
   228→      for (const f of IDENTITY_FIELDS) body[f.key] = identityValue(f.key)
   229→      const res = await fetch('/api/admin/settings', {
   230→        method: 'PUT',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify(body),
   233→      })
   234→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   235→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan')
   236→      toast({ title: 'Identitas dashboard tersimpan' })
   237→      setIdDraft({})
   238→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   239→      await refreshStatus()
   240→      setStep(2)
   241→    } catch (err) {
   242→      toast({ title: 'Gagal menyimpan identitas', description: String(err), variant: 'destructive' })
   243→    } finally {
   244→      setIdSaving(false)
   245→    }
   246→  }
   247→
   248→  /** Ganti password akun sendiri, lalu lanjut ke langkah berikutnya. */
   249→  async function changePassword() {
   250→    if (pwSaving) return
   251→    setPwError(null)
   252→    if (!pwCurrent || !pwNew || !pwConfirm) {
   253→      setPwError('Semua kolom password wajib diisi (atau kosongkan semua untuk melewati).')
   254→      return
   255→    }
   256→    if (pwNew.length < 8) {
   257→      setPwError('Password baru minimal 8 karakter.')
   258→      return
   259→    }
   260→    if (pwNew !== pwConfirm) {
   261→      setPwError('Konfirmasi password tidak sama dengan password baru.')
   262→      return
   263→    }
   264→    if (pwNew === pwCurrent) {
   265→      setPwError('Password baru harus berbeda dari password saat ini.')
   266→      return
   267→    }
   268→    setPwSaving(true)
   269→    try {
   270→      const res = await fetch('/api/admin/users/change-password', {
   271→        method: 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify({ currentPassword: pwCurrent, newPassword: pwNew }),
   274→      })
   275→      const json = (await res.json()) as { data?: { username: string }; error?: string }
   276→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal mengganti password')
   277→      toast({ title: 'Password berhasil diganti' })
   278→      setPwDone(true)
   279→      setPwCurrent('')
   280→      setPwNew('')
   281→      setPwConfirm('')
   282→      await refreshStatus()
   283→      setStep(3)
   284→    } catch (err) {
   285→      setPwError(String(err))
   286→    } finally {
   287→      setPwSaving(false)
   288→    }
   289→  }
   290→
   291→  /** Simpan konfigurasi AI Copilot, lalu lanjut ke langkah selesai. */
   292→  async function saveCopilot() {
   293→    if (cpSaving) return
   294→    if (cpIsDefault) {
   295→      setStep(4)
   296→      return
   297→    }
   298→    if (!cpModel.trim()) {
   299→      toast({ title: 'Nama model wajib diisi', variant: 'destructive' })
   300→      return
   301→    }
   302→    setCpSaving(true)
   303→    try {
   304→      const res = await fetch('/api/admin/settings/copilot', {
   305→        method: 'PUT',
   306→        headers: { 'Content-Type': 'application/json' },
   307→        body: JSON.stringify({
   308→          provider: cpProvider,
   309→          baseUrl: cpBase,
   310→          model: cpModel,
   311→          ...(cpKey.trim() ? { apiKey: cpKey.trim() } : {}),
   312→        }),
   313→      })
   314→      const json = (await res.json()) as { data?: CopilotSettingsDto; error?: string }
   315→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan')
   316→      toast({ title: 'Konfigurasi AI Copilot tersimpan' })
   317→      setCpKey('')
   318→      setCpTest(null)
   319→      await queryClient.invalidateQueries({ queryKey: ['copilot-settings'] })
   320→      await refreshStatus()
   321→      setStep(4)
   322→    } catch (err) {
   323→      toast({ title: 'Gagal menyimpan konfigurasi', description: String(err), variant: 'destructive' })
   324→    } finally {
   325→      setCpSaving(false)
   326→    }
   327→  }
   328→
   329→  /** Uji koneksi AI Copilot memakai nilai draf (key kosong → pakai tersimpan). */
   330→  async function testCopilot() {
   331→    setCpTesting(true)
   332→    setCpTest(null)
   333→    try {
   334→      const res = await fetch('/api/admin/settings/copilot/test', {
   335→        method: 'POST',
   336→        headers: { 'Content-Type': 'application/json' },
   337→        body: JSON.stringify({
   338→          provider: cpProvider,
   339→          ...(cpIsDefault
   340→            ? {}
   341→            : { baseUrl: cpBase, model: cpModel, ...(cpKey.trim() ? { apiKey: cpKey.trim() } : {}) }),
   342→        }),
   343→      })
   344→      const json = (await res.json()) as {
   345→        data?: { engine: string; reply: string; latencyMs: number }
   346→        error?: string
   347→      }
   348→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menguji koneksi')
   349→      setCpTest({
   350→        ok: true,
   351→        message: `Berhasil terhubung ke ${json.data.engine} dalam ${json.data.latencyMs} ms — balasan: "${json.data.reply.slice(0, 80)}"`,
   352→      })
   353→    } catch (err) {
   354→      setCpTest({ ok: false, message: String(err) })
   355→    } finally {
   356→      setCpTesting(false)
   357→    }
   358→  }
   359→
   360→  /** Tandai setup selesai dan tutup wizard. */
   361→  async function finishSetup() {
   362→    if (finishing) return
   363→    setFinishing(true)
   364→    try {
   365→      const res = await fetch('/api/admin/setup-wizard', { method: 'POST' })
   366→      const json = (await res.json()) as { data?: SetupWizardStatusDto; error?: string }
   367→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menandai setup selesai')
   368→      toast({ title: 'Setup selesai — dashboard siap digunakan' })
   369→      await refreshStatus()
   370→      onOpenChange(false)
   371→    } catch (err) {
   372→      toast({ title: 'Gagal menandai setup selesai', description: String(err), variant: 'destructive' })
   373→    } finally {
   374→      setFinishing(false)
   375→    }
   376→  }
   377→
   378→  /** Pergi ke langkah tertentu; refresh status saat memasuki langkah Selesai. */
   379→  async function goTo(next: number) {
   380→    setStep(next)
   381→    if (next === 4) await refreshStatus()
   382→  }
   383→
   384→  const pwFormActive = !!(pwCurrent || pwNew || pwConfirm)
   385→
   386→  // ---- Aksi utama per langkah ----
   387→  let primaryAction: { label: string; icon: React.ReactNode; onClick: () => void; disabled?: boolean; loading?: boolean }
   388→  let secondaryAction: { label: string; onClick: () => void } | null = null
   389→
   390→  if (step === 0) {
   391→    primaryAction = { label: 'Mulai Konfigurasi', icon: <Wand2 className="h-4 w-4" aria-hidden="true" />, onClick: () => goTo(1) }
   392→  } else if (step === 1) {
   393→    primaryAction = {
   394→      label: 'Simpan & Lanjut',
   395→      icon: idSaving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <Save className="h-4 w-4" aria-hidden="true" />,
   396→      onClick: saveIdentity,
   397→      disabled: idSaving || settingsQuery.isLoading,
   398→    }
   399→    secondaryAction = { label: 'Lewati', onClick: () => goTo(2) }
   400→  } else if (step === 2) {
   401→    primaryAction = {
   402→      label: pwFormActive ? 'Ganti Password & Lanjut' : 'Lanjut',
   403→      icon: pwSaving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <ShieldCheck className="h-4 w-4" aria-hidden="true" />,
   404→      onClick: pwFormActive ? changePassword : () => goTo(3),
   405→      disabled: pwSaving,
   406→    }
   407→  } else if (step === 3) {
   408→    primaryAction = {
   409→      label: cpIsDefault ? 'Lanjut' : 'Simpan & Lanjut',
   410→      icon: cpSaving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <Bot className="h-4 w-4" aria-hidden="true" />,
   411→      onClick: saveCopilot,
   412→      disabled: cpSaving || (!cpIsDefault && !cpModel.trim()),
   413→    }
   414→    if (!cpIsDefault) secondaryAction = { label: 'Lewati', onClick: () => goTo(4) }
   415→  } else {
   416→    primaryAction = {
   417→      label: 'Selesaikan Setup',
   418→      icon: finishing ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <CheckCircle2 className="h-4 w-4" aria-hidden="true" />,
   419→      onClick: finishSetup,
   420→      disabled: finishing,
   421→    }
   422→    secondaryAction = { label: 'Tutup tanpa menandai selesai', onClick: () => onOpenChange(false) }
   423→  }
   424→
   425→  return (
   426→    <Dialog open={open} onOpenChange={onOpenChange}>
   427→      <DialogContent className="flex max-h-[90dvh] w-full max-w-[min(42rem,calc(100%-2rem))] flex-col gap-0 overflow-hidden p-0 [&_[data-slot=dialog-close]]:rounded-md [&_[data-slot=dialog-close]]:text-white/80 [&_[data-slot=dialog-close]]:transition-colors [&_[data-slot=dialog-close]]:hover:bg-white/10 [&_[data-slot=dialog-close]]:hover:text-white">
   428→        {/* Header */}
   429→        <div className="bg-gradient-to-r from-[#17408b] to-[#12326e] px-5 py-4">
   430→          <DialogTitle className="flex items-center gap-2 text-base font-bold text-white">
   431→            <Wand2 className="h-5 w-5" aria-hidden="true" />
   432→            Setup Wizard — Konfigurasi Dashboard
   433→          </DialogTitle>
   434→          <DialogDescription className="mt-1 text-xs text-white/80">
   435→            Panduan berlangkah untuk menyiapkan identitas, keamanan, dan AI Copilot
   436→            dashboard keuangan daerah Anda.
   437→          </DialogDescription>
   438→        </div>
   439→
   440→        {/* Stepper */}
   441→        <nav aria-label="Progres langkah setup" className="border-b bg-muted/30 px-5 py-3">
   442→          <ol className="flex w-full items-start">
   443→            {STEPS.map((s, i) => {
   444→              const done = i < step
   445→              const current = i === step
   446→              return (
   447→                <li key={s.id} className="flex min-w-0 flex-1 flex-col items-center">
   448→                  <div className="flex w-full items-center">
   449→                    <span
   450→                      aria-hidden="true"
   451→                      className={`h-0.5 flex-1 rounded-full ${i === 0 ? 'invisible' : done || current ? 'bg-[#17408b]' : 'bg-border'}`}
   452→                    />
   453→                    <span
   454→                      aria-current={current ? 'step' : undefined}
   455→                      className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors ${
   456→                        done
   457→                          ? 'border-emerald-600 bg-emerald-600 text-white'
   458→                          : current
   459→                            ? 'border-[#17408b] bg-[#17408b] text-white'
   460→                            : 'border-border bg-background text-muted-foreground'
   461→                      }`}
   462→                    >
   463→                      {done ? (
   464→                        <Check className="h-4 w-4" aria-hidden="true" />
   465→                      ) : (
   466→                        <s.icon className="h-4 w-4" aria-hidden="true" />
   467→                      )}
   468→                    </span>
   469→                    <span
   470→                      aria-hidden="true"
   471→                      className={`h-0.5 flex-1 rounded-full ${i === STEPS.length - 1 ? 'invisible' : done ? 'bg-[#17408b]' : 'bg-border'}`}
   472→                    />
   473→                  </div>
   474→                  <span
   475→                    className={`mt-1 hidden whitespace-nowrap text-[10px] font-semibold sm:block ${
   476→                      current ? 'text-[#17408b]' : done ? 'text-emerald-700' : 'text-muted-foreground'
   477→                    }`}
   478→                  >
   479→                    {s.label}
   480→                  </span>
   481→                </li>
   482→              )
   483→            })}
   484→          </ol>
   485→        </nav>
   486→
   487→        {/* Isi langkah */}
   488→        <div className="nice-scrollbar min-h-[16rem] flex-1 overflow-y-auto px-5 py-5">
   489→          {/* ===== Langkah 0: Pengantar ===== */}
   490→          {step === 0 && (
   491→            <div>
   492→              <div className="mb-4 flex items-start gap-3">
   493→                <span className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-[#17408b]/10 text-[#17408b]">
   494→                  <Wand2 className="h-6 w-6" aria-hidden="true" />
   495→                </span>
   496→                <div>
   497→                  <h3 className="text-sm font-bold text-foreground">Selamat datang di Setup Wizard</h3>
   498→                  <p className="mt-0.5 text-xs text-muted-foreground">
   499→                    Wizard ini membantu menyiapkan hal-hal penting sebelum dashboard
   500→                    dipakai sepenuhnya. Semua langkah bersifat opsional dan dapat
   501→                    diubah lagi nanti lewat menu Pengaturan.
   502→                  </p>
   503→                </div>
   504→              </div>
   505→
   506→              {statusQuery.isLoading ? (

   951→/** Baris status dengan ikon + label + deskripsi (dipakai langkah Pengantar & Selesai). */
   952→function StatusRow({
   953→  icon,
   954→  tone,
   955→  label,
   956→  desc,
   957→  badge,
   958→}: {
   959→  icon: React.ReactNode
   960→  tone: 'ok' | 'warn' | 'muted'
   961→  label: string
   962→  desc: string
   963→  badge?: string
   964→}) {
   965→  const toneClass =
   966→    tone === 'ok'
   967→      ? 'border-emerald-200 bg-emerald-50'
   968→      : tone === 'warn'
   969→        ? 'border-amber-200 bg-amber-50'
   970→        : 'border-border bg-muted/40'
   971→  const iconClass =
   972→    tone === 'ok' ? 'text-emerald-700' : tone === 'warn' ? 'text-amber-600' : 'text-muted-foreground'
   973→
   974→  return (
   975→    <li className={`flex items-start gap-3 rounded-md border p-3 ${toneClass}`}>
   976→      <span className={`mt-0.5 shrink-0 ${iconClass}`}>{icon}</span>
   977→      <div className="min-w-0">
   978→        <p className="flex flex-wrap items-center gap-2 text-xs font-bold text-foreground">
   979→          {label}
   980→          {badge && (
   981→            <span className="rounded-full border border-border bg-background px-2 py-0.5 text-[10px] font-semibold text-muted-foreground">
   982→              {badge}
   983→            </span>
   984→          )}
   985→        </p>
   986→        <p className="mt-0.5 text-xs text-muted-foreground">{desc}</p>
   987→      </div>
   988→    </li>
   989→  )
   990→}
   991→
