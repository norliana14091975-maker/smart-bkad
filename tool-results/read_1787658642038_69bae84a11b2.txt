   262→  async function saveHeaderColor(value: string) {
   263→    const color = value.trim()
   264→    if (color !== '' && !/^#[0-9a-fA-F]{6}$/.test(color)) {
   265→      toast({ title: 'Warna tidak valid', description: 'Gunakan kode hex #rrggbb.', variant: 'destructive' })
   266→      return
   267→    }
   268→    try {
   269→      const res = await fetch('/api/admin/settings', {
   270→        method: 'PUT',
   271→        headers: { 'Content-Type': 'application/json' },
   272→        body: JSON.stringify({ headerColor: color }),
   273→      })
   274→      const json = (await res.json()) as { data?: AppSettingsDto; error?: string }
   275→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal menyimpan warna')

   277→      toast({ title: color ? 'Warna header diperbarui' : 'Warna header kembali ke gradien bawaan' })
   278→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   279→    } catch (err) {
   280→      toast({ title: 'Gagal menyimpan warna', description: String(err), variant: 'destructive' })
   281→    }

   316→  async function uploadImage(kind: 'logo' | 'sidebar-logo' | 'emblem' | 'favicon', file: File) {
   317→    const ref =
   318→      kind === 'logo'
   319→        ? logoInputRef
   320→        : kind === 'sidebar-logo'
   321→          ? sidebarLogoInputRef
   322→          : kind === 'emblem'
   323→            ? emblemInputRef
   324→            : faviconInputRef
   325→    if (ref) {
   326→      // reset nilai input agar file bernama sama bisa diunggah ulang
   327→      if (ref.current) ref.current.value = ''
   328→    }
   329→    if (!file.type.startsWith('image/')) {
   330→      toast({ title: 'File bukan gambar', description: 'Pilih file PNG/JPG/GIF/WebP/SVG/ICO.', variant: 'destructive' })
   331→      return
   332→    }
   333→    if (file.size > 2 * 1024 * 1024) {
   334→      toast({ title: 'File terlalu besar', description: 'Ukuran maksimum 2 MB.', variant: 'destructive' })
   335→      return
   336→    }
   337→    setUploading(kind)
   338→    try {
   339→      const fd = new FormData()
   340→      fd.append('file', file)
   341→      const res = await fetch(`/api/admin/settings/${kind}`, { method: 'POST', body: fd })
   342→      const json = (await res.json()) as {
   343→        data?: { logoUrl?: string; sidebarLogoUrl?: string; emblemUrl?: string; faviconUrl?: string }
   344→        error?: string
   345→      }
   346→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal mengunggah')
   347→      toast({
   348→        title:
   349→          kind === 'logo'
   350→            ? 'Logo diperbarui'
   351→            : kind === 'sidebar-logo'
   352→              ? 'Logo pojok kiri diperbarui'
   353→              : kind === 'emblem'
   354→                ? 'Lencana pojok kanan diperbarui'
   355→                : 'Favicon diperbarui',
   356→      })
   357→      await queryClient.invalidateQueries({ queryKey: ['settings'] })
   358→    } catch (err) {
   359→      toast({ title: 'Gagal mengunggah', description: String(err), variant: 'destructive' })
   360→    } finally {
   361→      setUploading(null)
   362→    }
   363→  }
