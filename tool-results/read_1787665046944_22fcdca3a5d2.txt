    96→    96→  return useQuery({
    97→    97→    queryKey: ['setup-wizard-status'],
    98→    98→    enabled,
    99→    99→    queryFn: async (): Promise<SetupWizardStatusDto> => {
   100→   100→      const res = await fetch('/api/admin/setup-wizard')
   101→   101→      if (!res.ok) throw new Error('Gagal memuat status Setup Wizard')
   102→   102→      const json = (await res.json()) as { data: SetupWizardStatusDto }
   103→   103→      return json.data
   104→   104→    },
   105→   105→    retry: 1,
   106→   106→  })

   249→   249→  async function changePassword() {

   251→   251→    setPwError(null)
   252→   252→    if (!pwCurrent || !pwNew || !pwConfirm) {
   253→   253→      setPwError('Semua kolom password wajib diisi (atau kosongkan semua untuk melewati).')

   255→   255→    }

   257→   257→      setPwError('Password baru minimal 8 karakter.')

   261→   261→      setPwError('Konfirmasi password tidak sama dengan password baru.')
   262→   262→      return

   264→   264→    if (pwNew === pwCurrent) {
   265→   265→      setPwError('Password baru harus berbeda dari password saat ini.')
   266→   266→      return
   267→   267→    }
   268→   268→    setPwSaving(true)
   269→   269→    try {
   270→   270→      const res = await fetch('/api/admin/users/change-password', {
   271→   271→        method: 'POST',
   272→   272→        headers: { 'Content-Type': 'application/json' },
   273→   273→        body: JSON.stringify({ currentPassword: pwCurrent, newPassword: pwNew }),
   274→   274→      })
   275→   275→      const json = (await res.json()) as { data?: { username: string }; error?: string }
   276→   276→      if (!res.ok || !json.data) throw new Error(json.error ?? 'Gagal mengganti password')
   277→   277→      toast({ title: 'Password berhasil diganti' })
   278→   278→      setPwDone(true)
   279→   279→      setPwCurrent('')
   280→   280→      setPwNew('')
   281→   281→      setPwConfirm('')
   282→   282→      await refreshStatus()
   283→   283→      setStep(3)
   284→   284→    } catch (err) {
   285→   285→      setPwError(String(err))
   286→   286→    } finally {
   287→   287→      setPwSaving(false)
   288→   288→    }
   289→   289→  }

   930→   930→            )}

   944→   944→          </div>
   945→   945→        </div>
   946→   946→      </DialogContent>
   947→   947→    </Dialog>
   948→   948→  )
   949→   949→}

   951→   951→/** Baris status dengan ikon + label + deskripsi (dipakai langkah Pengantar & Selesai). */

   953→   953→  icon,
   954→   954→  tone,
   955→   955→  label,
   956→   956→  desc,
   957→   957→  badge,
   958→   958→}: {
   959→   959→  icon: React.ReactNode
   960→   960→  tone: 'ok' | 'warn' | 'muted'
   961→   961→  label: string
   962→   962→  desc: string
   963→   963→  badge?: string

   965→   965→  const toneClass =
   966→   966→    tone === 'ok'

   971→   971→  const iconClass =
   972→   972→    tone === 'ok' ? 'text-emerald-700' : tone === 'warn' ? 'text-amber-600' : 'text-muted-foreground'
   973→   973→
   974→   974→  return (
   975→   975→    <li className={`flex items-start gap-3 rounded-md border p-3 ${toneClass}`}>
   976→   976→      <span className={`mt-0.5 shrink-0 ${iconClass}`}>{icon}</span>

   978→   978→        <p className="flex flex-wrap items-center gap-2 text-xs font-bold text-foreground">
   979→   979→          {label}
   980→   980→          {badge && (
   981→   981→            <span className="rounded-full border border-border bg-background px-2 py-0.5 text-[10px] font-semibold text-muted-foreground">
   982→   982→              {badge}
   983→   983→            </span>
   984→   984→          )}
   985→   985→        </p>

   987→   987→      </div>
   988→   988→    </li>
   989→   989→  )
   990→   990→}
